URR — последние исправления

Устранение «зависаний» после добавления busy в UART-RX. Четыре файла: urr_uart_rx.v, urr_rdy_ctrl_simple.v, urr_crc.v, urr.v. Добавленные строки подсвечены зелёным, изменённые — жёлтым.

Навигация

  • Сводка причин
  • urr_uart_rx.v +5
  • urr_rdy_ctrl_simple.v +5
  • urr_crc.v +30
  • urr.v +2

Сводка причин зависания

Добавление busy в приёмник UART обнажило три архитектурные проблемы.

1. Синхронизатор в RX замораживался на всё время передачи. После снятия busy фронт мог детектироваться посреди чужого байта → мусорные данные / отсутствие rx_valid.
2. urr_rdy_ctrl_simple восстанавливал rdy только по vd_o (т.е. по done). При таймауте/битом ответе done не приходил → rdy_r навсегда оставался в 0 → все следующие en_i игнорировались.
3. urr_crc не имел таймаутов на WAIT_RX / WAIT_RX_255 / COLLECT — при потере байта crc_running залипал в 1 и блокировал start для urr_osn.

urr_uart_rx.v

Синхронизатор работает всегда; глушится только детектор фронта (с 1 тактом запаса после снятия busy).

urr_rdy_ctrl_simple.v

Новый вход err_o: ошибка тоже восстанавливает rdy_r.

urr_crc.v

Таймауты в WAIT_RX/WAIT_RX_255/COLLECT. В ERROR_ST импульс crc_done.

urr.v

Проброс .err_o(err_freq_from_urr) в rdy_ctrl_simple. Include test_urr.v.

urr_uart_rx.v

Синхронизатор больше не замораживается в idle — всегда отслеживает линию. Фронт глушится через busy || busy_d (один такт запаса), чтобы не поймать уже идущий байт slave'а сразу после снятия busy.

добавлено изменено без изменений 
module urr_uart_rx (
    input  wire       clk,
    input  wire       rst_n,
   (* DONT_TOUCH = "yes" *)  (* IOB = "TRUE" *) input  wire       rx,
   (* DONT_TOUCH = "yes" *)  (* IOB = "TRUE" *) input  wire       busy,
   (* DONT_TOUCH = "yes" *)  output reg  [7:0] data,
   (* DONT_TOUCH = "yes" *)  output wire       valid,
   (* DONT_TOUCH = "yes" *)  output reg        error
);

    localparam BAUD_RATE = 115200;
    localparam CLK_FREQ  = 60_000_000;
    localparam BAUD_CNT  = CLK_FREQ / BAUD_RATE;

   (* DONT_TOUCH = "yes" *)  reg [15:0] baud_cnt;
   (* DONT_TOUCH = "yes" *)  reg [3:0]  bit_cnt;
   (* DONT_TOUCH = "yes" *)  reg [8:0]  shift_reg;
   (* DONT_TOUCH = "yes" *)  reg rx_sync1, rx_sync2;
   (* DONT_TOUCH = "yes" *)  reg rx_falling;
   (* DONT_TOUCH = "yes" *)  reg receiving;
   (* DONT_TOUCH = "yes" *)  (* SRL_STYLE = "register" *) reg valid_reg;
   (* DONT_TOUCH = "yes" *)  reg error_reg;
   (* DONT_TOUCH = "yes" *)  reg parity;
   (* DONT_TOUCH = "yes" *)  reg fft_rx;
   (* DONT_TOUCH = "yes" *)  reg busy_d;   // NEW: задержанный busy для settling

    assign valid = valid_reg;

    // Первый триггер синхронизатора (IOB)
    always @(posedge clk) begin
        fft_rx <= rx;
    end

    // NEW: задержанный busy — используется для settling после снятия TX
    always @(posedge clk) begin
        if (rst_n) busy_d <= 1'b0;
        else       busy_d <= busy;
    end

    // MOD: синхронизатор работает ВСЕГДА (не замораживается),
    //      детектор фронта заглушается только через busy || busy_d
    always @(posedge clk) begin
        if (rst_n) begin
            rx_sync1   <= 1'b1;
            rx_sync2   <= 1'b1;
            rx_falling <= 1'b0;
        end else begin
            rx_sync1 <= fft_rx;
            rx_sync2 <= rx_sync1;

            // Фронт детектируем только когда TX точно завершён + 1 такт запаса
            if (busy || busy_d) begin
                rx_falling <= 1'b0;
            end else begin
                rx_falling <= rx_sync2 & ~rx_sync1;   // 1→0
            end
        end
    end

    // Приёмник байта (без изменений)
    always @(posedge clk) begin
        if (rst_n) begin
            receiving <= 0; baud_cnt <= 0; bit_cnt <= 0;
            shift_reg <= 0; valid_reg <= 0; error_reg <= 0;
            data <= 0; parity <= 1;
        end else begin
            valid_reg <= 0;
            if (!receiving) begin
                if (rx_falling) begin
                    receiving <= 1; baud_cnt <= 0; bit_cnt <= 0;
                end
            end else begin
                baud_cnt <= baud_cnt + 1;
                if (baud_cnt == BAUD_CNT/2) begin
                    if (bit_cnt >= 1 && bit_cnt <= 8)
                        shift_reg <= {rx_sync2, shift_reg[8:1]};
                    else if (bit_cnt == 10) begin
                        if (rx_sync2 == 1) begin
                            data <= shift_reg[8:1]; valid_reg <= 1;
                        end else error_reg <= 1;
                        receiving <= 0;
                    end
                    bit_cnt <= bit_cnt + 1;
                end
                if (baud_cnt == BAUD_CNT) baud_cnt <= 0;
            end
        end
    end
endmodule

urr_rdy_ctrl_simple.v

Новый вход err_o от urr_osn. При err_rise rdy_r восстанавливается в 1, как и при vd_rise. Без этого фикса любая ошибка (таймаут, битый ответ) навсегда блокировала en_i.

добавлено изменено 
module urr_rdy_ctrl_simple (
    input  wire clk,
    input  wire rst_i,

    // Внешние сигналы
    (* DONT_TOUCH = "yes" *) input  wire urr_connected_i,
    (* DONT_TOUCH = "yes" *) input  wire en_i,
    (* DONT_TOUCH = "yes" *) input  wire crc_check_i,

    // От рабочих модулей
    (* DONT_TOUCH = "yes" *) input  wire vd_o,
    (* DONT_TOUCH = "yes" *) input  wire vd_crc_o,
    (* DONT_TOUCH = "yes" *) input  wire crc_match_i,
    (* DONT_TOUCH = "yes" *) input  wire err_o,          // NEW: error от urr_osn

    // Выходы
    (* DONT_TOUCH = "yes" *) output wire rdy_o,
    (* DONT_TOUCH = "yes" *) output wire err_freq_code_latch_o,
    (* DONT_TOUCH = "yes" *) output wire crc_result_latch_o,
    (* DONT_TOUCH = "yes" *) output wire vd_o_force_o,
    (* DONT_TOUCH = "yes" *) output wire vd_crc_o_force_o
);

    // ---------- Детекторы фронтов ----------
    (* DONT_TOUCH = "yes" *) reg en_d, crc_d, vd_d, vd_crc_d, conn_d, conn_dd, err_d;

    always @(posedge clk) begin
        if (rst_i) begin
            en_d <= 1'b0; crc_d <= 1'b0; vd_d <= 1'b0;
            vd_crc_d <= 1'b0; conn_d <= 1'b0; conn_dd <= 1'b0;
            err_d <= 1'b0;
        end else begin
            en_d <= en_i;      crc_d <= crc_check_i;
            vd_d <= vd_o;      vd_crc_d <= vd_crc_o;
            conn_d <= urr_connected_i;  conn_dd <= conn_d;
            err_d <= err_o;
        end
    end

    (* DONT_TOUCH = "yes" *) wire en_rise     = en_i        & ~en_d;
    (* DONT_TOUCH = "yes" *) wire crc_rise    = crc_check_i & ~crc_d;
    (* DONT_TOUCH = "yes" *) wire vd_rise     = vd_o        & ~vd_d;
    (* DONT_TOUCH = "yes" *) wire vd_crc_rise = vd_crc_o    & ~vd_crc_d;
    (* DONT_TOUCH = "yes" *) wire err_rise    = err_o       & ~err_d;   // NEW
    (* DONT_TOUCH = "yes" *) wire conn_rise   = conn_d      & ~conn_dd;
    (* DONT_TOUCH = "yes" *) wire conn_fall   = ~conn_d      & conn_dd;

    // ---------- Защёлки состояния ----------
    (* DONT_TOUCH = "yes" *) reg rdy_r, op_type_r, err_freq_r, crc_result_r;
    (* DONT_TOUCH = "yes" *) reg vd_force_r, vd_crc_force_r;

    always @(posedge clk) begin
        if (rst_i) begin
            rdy_r <= 1'b0; op_type_r <= 1'b0;
            err_freq_r <= 1'b0; crc_result_r <= 1'b0;
            vd_force_r <= 1'b0; vd_crc_force_r <= 1'b0;
        end else begin
            vd_force_r <= 1'b0;
            vd_crc_force_r <= 1'b0;

            // 1) Подключение URR: rdy=1, сброс защёлок ошибки
            if (conn_rise) begin
                rdy_r <= 1'b1;
                err_freq_r <= 1'b0;
                crc_result_r <= 1'b0;
            end

            // 2) Отключение URR: rdy=0; при работе — защёлка ошибки + форс vd
            if (conn_fall) begin
                rdy_r <= 1'b0;
                if (rdy_r == 1'b0) begin
                    if (op_type_r == 1'b0) begin
                        err_freq_r <= 1'b1; vd_force_r <= 1'b1;
                    end else begin
                        crc_result_r <= 1'b1; vd_crc_force_r <= 1'b1;
                    end
                end
            end

            // 3) Старт операции — только если rdy=1
            if (rdy_r == 1'b1) begin
                if (en_rise) begin rdy_r <= 1'b0; op_type_r <= 1'b0; end
                else if (crc_rise) begin rdy_r <= 1'b0; op_type_r <= 1'b1; end
            end

            // 4) Завершение — только если URR ещё подключён.
            //    MOD: по err_rise тоже восстанавливаем rdy,
            //         иначе система навсегда встанет в ожидании vd_o.
            if (rdy_r == 1'b0 && urr_connected_i) begin
                if ((vd_rise || err_rise) && op_type_r == 1'b0) rdy_r <= 1'b1;
                if ( vd_crc_rise           && op_type_r == 1'b1) rdy_r <= 1'b1;
            end
        end
    end

    assign rdy_o                 = rdy_r;
    assign err_freq_code_latch_o = err_freq_r;
    assign crc_result_latch_o    = crc_result_r;
    assign vd_o_force_o          = vd_force_r;
    assign vd_crc_o_force_o      = vd_crc_force_r;

endmodule

urr_crc.v

Добавлены таймауты в WAIT_RX, WAIT_RX_255, COLLECT. В ERROR_ST теперь импульсит crc_done, чтобы внешний rdy_ctrl мог корректно завершить CRC-операцию.

добавлено изменено 
module urr_crc (
    input  wire        clk,
    input  wire        rst_n,
   (* DONT_TOUCH = "yes" *)input  wire        start_crc,
   (* DONT_TOUCH = "yes" *) input  wire        rx_valid,
   (* DONT_TOUCH = "yes" *) input  wire [7:0]  rx_data,
   (* DONT_TOUCH = "yes" *) input  wire        rx_error,
   (* DONT_TOUCH = "yes" *) input  wire        tx_busy,
   (* DONT_TOUCH = "yes" *) input  wire        urr_error,
   (* DONT_TOUCH = "yes" *) output reg         start_urr,
   (* DONT_TOUCH = "yes" *) output reg  [7:0]  cmd_urr,
   (* DONT_TOUCH = "yes" *) output reg  [31:0] crc_result,
   (* DONT_TOUCH = "yes" *) output reg         crc_done,
   (* DONT_TOUCH = "yes" *) output wire        crc_running,
   (* DONT_TOUCH = "yes" *) output reg         crc_match
);

    localparam IDLE=4'd0, SEND=4'd1, WAIT_RX=4'd2, COLLECT=4'd3,
               SEND_255=4'd4, WAIT_RX_255=4'd5, COMPARE=4'd6,
               DONE_ST=4'd7, ERROR_ST=4'd8;
    localparam CRC_TIMEOUT_LIMIT = 500_000;   // NEW ~8ms @60MHz
    localparam COLLECT_TIMEOUT   = 200_000;   // NEW на сбор FIFO

    (* dont_touch = "true" *) (* keep = "true" *)
    (* fsm_encoding = "one_hot" *) (* use_clock_enable = "false" *)
    reg [3:0] state, next_state;

   (* DONT_TOUCH = "yes" *) reg [5:0]  cmd_cnt;
   (* DONT_TOUCH = "yes" *) reg [1:0]  byte_cnt;
   (* DONT_TOUCH = "yes" *) reg [7:0]  read_cnt;
   (* DONT_TOUCH = "yes" *) reg [31:0] crc_reg;
    wire [31:0] crc_out;
    wire [5:0] cmd_cnt_d;

   (* DONT_TOUCH = "yes" *) reg [1:0]  byte_cnt_255;
   (* DONT_TOUCH = "yes" *) reg [31:0] crc_received;
   (* DONT_TOUCH = "yes" *) reg [31:0] crc_timeout;   // NEW

   (* DONT_TOUCH = "yes" *) wire        fifo_wr_en;
   (* DONT_TOUCH = "yes" *) wire [7:0]  fifo_wr_data;
   (* DONT_TOUCH = "yes" *) wire        fifo_rd_en;
   (* DONT_TOUCH = "yes" *) wire [7:0]  fifo_rd_data;
   (* DONT_TOUCH = "yes" *) wire        fifo_empty, fifo_full;
   (* DONT_TOUCH = "yes" *) wire [7:0]  fifo_count;

   (* DONT_TOUCH = "yes" *) reg        fifo_rd_en_st;
   (* DONT_TOUCH = "yes" *) reg        crc_update_pending;
   (* DONT_TOUCH = "yes" *) reg [7:0]  fifo_rd_data_reg;
   (* DONT_TOUCH = "yes" *) reg [21:0] send_delay;

    urr_fifo_200 #(.DATA_WIDTH(8), .DEPTH(200)) u_fifo (
        .clk(clk), .rst(rst_n),
        .wr_en(fifo_wr_en), .wr_data(fifo_wr_data),
        .rd_en(fifo_rd_en), .rd_data(fifo_rd_data),
        .full(fifo_full), .empty(fifo_empty), .count(fifo_count)
    );

    urr_crc_wrapper u_crc_wrap (
        .enable(fifo_rd_en_st), .crcIn(crc_reg),
        .data(fifo_rd_data), .crcOut(crc_out)
    );

    // ---- Счётчик команд ----
    always @(posedge clk) begin
        if (rst_n) cmd_cnt <= 6'd1;
        else begin
            if (state == IDLE || state == DONE_ST) cmd_cnt <= 6'd1;
            else if (state == WAIT_RX && rx_valid && byte_cnt == 2'd3 && cmd_cnt < 6'd50)
                cmd_cnt <= cmd_cnt + 1;
        end
    end

    // NEW: таймаут-счётчик с автосбросом при любом валидном событии
    always @(posedge clk) begin
        if (rst_n) crc_timeout <= 32'd0;
        else begin
            if (state == IDLE || state == DONE_ST || rx_valid ||
                (state == COLLECT && read_cnt == 8'd200)) begin
                crc_timeout <= 32'd0;
            end else if (crc_timeout < CRC_TIMEOUT_LIMIT) begin
                crc_timeout <= crc_timeout + 1'b1;
            end
        end
    end

    // ---- Комбинационная логика переходов ----
     (* full_case, parallel_case *)
    always @(*) begin
        next_state = state;
        case (state)
            IDLE: if (start_crc) next_state = SEND;

            SEND: if (!tx_busy && send_delay == 15000) next_state = WAIT_RX;

            WAIT_RX: begin
                if (rx_valid) begin
                    if (byte_cnt == 2'd3) begin
                        if (cmd_cnt == 6'd50) next_state = COLLECT;
                        else next_state = SEND;
                    end else next_state = WAIT_RX;
                end
                else if (crc_timeout >= CRC_TIMEOUT_LIMIT)
                    next_state = ERROR_ST;                 // NEW
                else next_state = WAIT_RX;
            end

            COLLECT: begin
                if (read_cnt == 8'd200) next_state = SEND_255;
                else if (crc_timeout >= COLLECT_TIMEOUT)
                    next_state = ERROR_ST;                 // NEW
                else next_state = COLLECT;
            end

            SEND_255: if (!tx_busy && send_delay == 15000) next_state = WAIT_RX_255;

            WAIT_RX_255: begin
                if (rx_valid) begin
                    if (byte_cnt_255 == 2'd3) next_state = COMPARE;
                    else next_state = WAIT_RX_255;
                end
                else if (crc_timeout >= CRC_TIMEOUT_LIMIT)
                    next_state = ERROR_ST;                 // NEW
                else next_state = WAIT_RX_255;
            end

            COMPARE: next_state = DONE_ST;
            DONE_ST: next_state = IDLE;
            ERROR_ST: next_state = IDLE;
            default: next_state = IDLE;
        endcase
    end

    // ---- Синхронная логика ----
    always @(posedge clk) begin
        if (rst_n) begin
            state <= IDLE; byte_cnt <= 2'd0; byte_cnt_255 <= 2'd0;
            read_cnt <= 8'd0; crc_reg <= 32'hF5F5F5F5;
            crc_received <= 32'd0; crc_result <= 32'd0;
            start_urr <= 1'b0; cmd_urr <= 8'd0;
            crc_done <= 1'b0; crc_match <= 1'b0;
            fifo_rd_en_st <= 1'b0; crc_update_pending <= 1'b0;
            fifo_rd_data_reg <= 8'd0; send_delay <= 22'd0;
        end else begin
            start_urr <= 1'b0; crc_done <= 1'b0; crc_match <= 1'b0;

            if (fifo_rd_en) begin
                fifo_rd_data_reg <= fifo_rd_data;
                crc_update_pending <= 1'b1;
                fifo_rd_en_st <= 1'b1;
            end

            if (next_state == SEND && state != SEND) send_delay <= 22'd0;
            if (next_state == SEND_255 && state != SEND_255) send_delay <= 22'd0;

            case (state)
                IDLE: if (start_crc) begin
                    byte_cnt <= 2'd0; byte_cnt_255 <= 2'd0;
                    read_cnt <= 8'd0; crc_reg <= 32'hF5F5F5F5;
                    crc_received <= 32'd0; fifo_rd_en_st <= 1'b0;
                    crc_update_pending <= 1'b0; send_delay <= 22'd0;
                end

                SEND: begin
                    if (send_delay < 15000) send_delay <= send_delay + 1;
                    if (send_delay == 15000 && !tx_busy) begin
                        start_urr <= 1'b1; cmd_urr <= cmd_cnt_d; byte_cnt <= 2'd0;
                    end
                end

                WAIT_RX: if (rx_valid) begin
                    if (byte_cnt == 2'd3) byte_cnt <= 2'd0;
                    else byte_cnt <= byte_cnt + 1;
                end

                COLLECT: begin
                    crc_reg <= 32'hF5F5F5F5;
                    $display("COLLECT: read_cnt=%0d, data=0x%02h, crc_in=0x%08h, crc_out=0x%08h",
                             read_cnt, fifo_rd_data, crc_reg, crc_out);
                    read_cnt <= read_cnt + 1;
                    crc_reg <= crc_out;
                    crc_update_pending <= 1'b0; send_delay <= 22'd0;
                end

                SEND_255: begin
                    if (send_delay < 15000) send_delay <= send_delay + 1;
                    fifo_rd_en_st <= 1'b0;
                    if (send_delay == 15000 && !tx_busy) begin
                        start_urr <= 1'b1; cmd_urr <= 8'hFF; byte_cnt_255 <= 2'd0;
                    end
                end

                WAIT_RX_255: if (rx_valid) begin
                    case (byte_cnt_255)
                        2'd0: crc_received[31:24] <= rx_data;
                        2'd1: crc_received[23:16] <= rx_data;
                        2'd2: crc_received[15:8]  <= rx_data;
                        2'd3: crc_received[7:0]   <= rx_data;
                    endcase
                    if (byte_cnt_255 < 2'd3) byte_cnt_255 <= byte_cnt_255 + 1;
                end

                COMPARE: begin
                    crc_result <= crc_reg ^ 32'hFFFFFFFF;
                    crc_match  <= !((crc_reg ^ 32'hFFFFFFFF) == crc_received);
                end

                DONE_ST: begin
                    crc_done <= 1'b1;
                    byte_cnt <= 2'd0; byte_cnt_255 <= 2'd0;
                    read_cnt <= 8'd0; fifo_rd_en_st <= 1'b0;
                    crc_update_pending <= 1'b0; send_delay <= 22'd0;
                end

                ERROR_ST: begin                        // MOD: теперь импульсит crc_done
                    crc_done <= 1'b1;                  // чтобы внешний rdy_ctrl завершил операцию
                    crc_match <= 1'b0;
                    byte_cnt <= 2'd0; byte_cnt_255 <= 2'd0;
                    read_cnt <= 8'd0; fifo_rd_en_st <= 1'b0;
                    crc_update_pending <= 1'b0; send_delay <= 22'd0;
                end
            endcase

            state <= next_state;
        end
    end

    assign fifo_wr_en   = (state == WAIT_RX) && rx_valid && !fifo_full;
    assign fifo_wr_data = rx_data;
    assign fifo_rd_en   = (state == COLLECT);
    assign crc_running  = (state != IDLE) && (state != DONE_ST) && (state != ERROR_ST);
    assign cmd_cnt_d    = cmd_cnt;

endmodule

    wire       err_freq_code_o, vd_crc_o, crc_result_o, rdy_o;
    reg        is_connected_o, is_con_1, is_con_2;
    wire       clk_60_i, rst;
    wire       in_rising = is_con_1 && !is_con_2;
    wire       in_descending = is_con_2 && !is_con_1;

    assign rst   = rst_i;   // активный-высокий от RES
    assign start = crc_running ? start_urr : (rdy_o ? en_i : 1'b0);
    assign cmd   = crc_running ? cmd_urr   : addr_i;
    assign crc_check_ii = rdy_o ? crc_check_i : 1'b0;

    RES res (.clk(clk_50_i), .rst(rst_i));

    clk_wiz_0 clk_wiz (
      .clk_out1(clk_60_i), .reset(rst_i), .locked(), .clk_in1(clk_50_i)
    );

    test_urr u_test (
     .clk(clk_60_i), .rst(rst_i),
     .urr_connected_i(urr_connected_i),
     .en_i(en_i), .addr_i(addr_i), .crc_check_i(crc_check_i)
    );

    urr_pulse_stretcher pulse_irq_con_o (
        .clk(clk_60_i), .rst(rst_i),
        .imp_i(urr_connected_i), .imp_out(irq_con_o)
    );
    urr_pulse_stretcher pulse_irq_o (
        .clk(clk_60_i), .rst(rst_i),
        .imp_i(vd_o), .imp_out(irq_o)
    );

    urr_uart_tx u_tx (
        .clk(clk_60_i), .rst_n(rst_i),
        .start(tx_start), .data(tx_data),
        .busy(tx_busy), .tx(rs_tx_o)
    );

    urr_uart_rx u_rx (
        .clk(clk_60_i), .rst_n(rst_i),
        .rx(rs_rx_i), .data(rx_data),
        .valid(rx_valid), .error(rx_error),
        .busy(tx_busy)                // NEW: проброс busy в RX
    );

    urr_crc u_urr_crc (
        .clk(clk_60_i), .rst_n(rst_i),
        .start_crc(crc_check_ii),
        .rx_valid(rx_valid), .rx_data(rx_data), .rx_error(rx_error),
        .tx_busy(tx_busy), .urr_error(err_freq_from_urr),
        .start_urr(start_urr), .cmd_urr(cmd_urr),
        .crc_result(crc_out_o), .crc_done(vd_urr_crc),
        .crc_running(crc_running), .crc_match(crc_result_from_urr_crc)
    );

    urr_osn u_urr_osn (
        .clk(clk_60_i), .rst_n(rst_i),
        .start(start), .cmd(cmd),
        .tx_busy(tx_busy), .tx_start(tx_start), .tx_data(tx_data),
        .rx_valid(rx_valid), .rx_data(rx_data), .rx_error(rx_error),
        .freq(freq), .done(vd_o_urr), .error(err_freq_from_urr),
        .dir(rs_dir_o)
    );

    assign err_freq_code_o = (err_freq_from_urr && !crc_running) || err_freq_code_latch;
    assign crc_result_o    = crc_result_from_urr_crc || crc_result_latch;
    assign vd_crc_o        = vd_urr_crc || vd_crc_o_force_o;
    assign vd_o            = vd_o_force_o || vd_o_urr;
    assign freq_code_o     = {8'h00, freq};

    urr_rdy_ctrl_simple rdy_ctrl_simple (
        .clk(clk_60_i), .rst_i(rst_i),
        .urr_connected_i(urr_connected_i),
        .en_i(en_i), .crc_check_i(crc_check_i),
        .vd_o(vd_o_urr), .vd_crc_o(vd_urr_crc),
        .crc_match_i(crc_result_from_urr_crc),
        .err_o(err_freq_from_urr),        // NEW: пробрасываем ошибку
        .rdy_o(rdy_o),
        .err_freq_code_latch_o(err_freq_code_latch),
        .crc_result_latch_o(crc_result_latch),
        .vd_o_force_o(vd_o_force_o),
        .vd_crc_o_force_o(vd_crc_o_force_o)
    );

    always @(posedge clk_60_i) begin
        if (rst_i) begin is_con_1 <= 1'b0; is_con_2 <= 1'b0; end
        else begin is_con_1 <= urr_connected_i; is_con_2 <= is_con_1; end
    end

    always @(posedge clk_60_i) begin
        if (rst_i) is_connected_o <= 1'b0;
        else if (in_rising || in_descending && !urr_connected_i) is_connected_o <= 1'b1;
        else if (in_rising || in_descending &&  urr_connected_i) is_connected_o <= 1'b0;
    end

endmodule
